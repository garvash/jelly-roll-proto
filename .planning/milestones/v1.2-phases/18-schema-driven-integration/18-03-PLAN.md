       ```python
       import json, tempfile, shutil
       def test_schema_mutation():
           """Changing tile_coords in schema changes lookup result (Success Criterion 3, D-17)."""
           # Copy real schema to temp file
           with open("assets/entity-schema.json") as f:
               data = json.load(f)
           # Modify: IntGrid 1 (solid) coords from [0,1] to [5,5]
           data["biomes"]["cavern"]["tile_coords"]["1"] = [5, 5]
           with tempfile.NamedTemporaryFile(mode='w', suffix='.json', delete=False) as tmp:
               json.dump(data, tmp)
               tmp_path = tmp.name
           try:
               schema.init(tmp_path)
               assert schema.get_val_to_tile()[1] == (5, 5), "Mutation not reflected"
           finally:
               os.unlink(tmp_path)
               # Re-init with real schema to not affect other tests
               schema.init("assets/entity-schema.json")
       ```
