    ```python
    """Verify gamepad controller support via _ACTION_MAP (D-04, D-05)."""
    import unittest
    import pyxel

    # Must init pyxel minimally for constants to be available
    # pyxel constants are module-level, no init needed

    from src.core.input import _ACTION_MAP

    class TestGamepadMapping(unittest.TestCase):
        def test_dpad_left(self):
            self.assertIn(pyxel.GAMEPAD1_BUTTON_DPAD_LEFT, _ACTION_MAP["left"])

        def test_dpad_right(self):
            self.assertIn(pyxel.GAMEPAD1_BUTTON_DPAD_RIGHT, _ACTION_MAP["right"])

        def test_dpad_up(self):
            self.assertIn(pyxel.GAMEPAD1_BUTTON_DPAD_UP, _ACTION_MAP["up"])

        def test_dpad_down(self):
            self.assertIn(pyxel.GAMEPAD1_BUTTON_DPAD_DOWN, _ACTION_MAP["down"])

        def test_button_a_jump(self):
            """A (bottom) = Jump per D-04."""
            self.assertIn(pyxel.GAMEPAD1_BUTTON_A, _ACTION_MAP["jump"])

        def test_button_b_spit(self):
            """B (right) = Spit per D-04."""
            self.assertIn(pyxel.GAMEPAD1_BUTTON_B, _ACTION_MAP["spit"])

        def test_button_x_dash(self):
            """X (left) = Dash per D-04."""
            self.assertIn(pyxel.GAMEPAD1_BUTTON_X, _ACTION_MAP["dash"])

        def test_keyboard_keys_preserved(self):
            """Existing keyboard bindings unchanged."""
            self.assertIn(pyxel.KEY_LEFT, _ACTION_MAP["left"])
            self.assertIn(pyxel.KEY_A, _ACTION_MAP["left"])
            self.assertIn(pyxel.KEY_SPACE, _ACTION_MAP["jump"])
            self.assertIn(pyxel.KEY_Z, _ACTION_MAP["spit"])
            self.assertIn(pyxel.KEY_V, _ACTION_MAP["dash"])
    ```
