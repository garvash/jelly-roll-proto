Create `src/core/input.py` with the following exact implementation per D-09:

```python
import pyxel

# Logical action -> list of physical keys (D-09)
_ACTION_MAP = {
    "left":   [pyxel.KEY_LEFT, pyxel.KEY_A],
    "right":  [pyxel.KEY_RIGHT, pyxel.KEY_D],
    "up":     [pyxel.KEY_UP, pyxel.KEY_W],
    "down":   [pyxel.KEY_DOWN, pyxel.KEY_S],
    "jump":   [pyxel.KEY_SPACE],
    "spit":   [pyxel.KEY_Z, pyxel.KEY_J],
    "dash":   [pyxel.KEY_V, pyxel.KEY_K],
}

_hold_frames = {}
_prev_hold_frames = {}

def btn(action):
    return any(pyxel.btn(k) for k in _ACTION_MAP[action])

def btnp(action, hold=None, repeat=None):
    kwargs = {}
    if hold is not None:
        kwargs["hold"] = hold
    if repeat is not None:
        kwargs["repeat"] = repeat
    return any(pyxel.btnp(k, **kwargs) for k in _ACTION_MAP[action])

def btnr(action):
    return any(pyxel.btnr(k) for k in _ACTION_MAP[action])

def update():
    global _prev_hold_frames
    _prev_hold_frames = dict(_hold_frames)
    for action, keys in _ACTION_MAP.items():
        if any(pyxel.btn(k) for k in keys):
            _hold_frames[action] = _hold_frames.get(action, 0) + 1
        else:
            _hold_frames[action] = 0

def hold_frames(action):
    return _hold_frames.get(action, 0)

def was_tap(action, threshold):
    if any(pyxel.btnr(k) for k in _ACTION_MAP[action]):
        prev = _prev_hold_frames.get(action, 0)
        return 0 < prev <= threshold
    return False
```

Key design notes:
- `_prev_hold_frames` stores the hold count from BEFORE `update()` resets it. This is critical for `was_tap()` -- on the release frame, `_hold_frames[action]` is still the held value because `update()` runs first, then `btnr` is checked. Actually: `update()` sets `_hold_frames[action] = 0` on the release frame (because `btn()` returns False). So `_prev_hold_frames` captures the value from the LAST frame where the key was held. This is exactly what `was_tap()` needs.
- `update()` must be called once per frame BEFORE any input checks. It will be called from `Player.update()` or `Game.update()`.
