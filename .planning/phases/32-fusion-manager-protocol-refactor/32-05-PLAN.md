       ```python
       def on_tick(self, player, slime, dt: float) -> TickResult:
           # 1. Re-clamp velocity (verbatim from apply_diving_physics).
           dy = tuning.DRILL_SPEED
           if input_manager.btn("left"):
               dx = -tuning.DRILL_DRIFT_SPEED
           elif input_manager.btn("right"):
               dx = tuning.DRILL_DRIFT_SPEED
           else:
               dx = 0.0

           # 2. Juice empty → exit (b)
           if slime.juice <= 0:
               return TickResult(dx=dx, dy=dy, request_exit=True, exit_reason="juice_empty")

           # 3. Block-break detection (per-frame, before move_and_collide runs)
           tile_coord = player.level_map.get_destructible_at(player.x, player.y, player.w, player.h)
           if tile_coord:
               tx, ty = tile_coord
               tile_type = player.level_map.get_tile(tx, ty)
               if player.game:
                   player.game.on_block_destroyed(tx, ty, tile_type)
               player.level_map.remove_tile(tx, ty)
               if player.game:
                   player.game.spawn_explosion(tx * tuning.TILE_SIZE, ty * tuning.TILE_SIZE,
                                                EXPLOSION_SIZE_PX)
               if tile_type == INTGRID_CRACKED_V:
                   slime.consume(tuning.DRILL_CRACKED_V_COST)
               else:
                   slime.refill(tuning.DRILL_BLOCK_REFUND)
               player.on_block_break()
               event_bus.emit("drill_block_break", tx=tx, ty=ty)
               return TickResult(dx=dx, dy=dy)  # drill continues, no exit

           # 4. Solid-landing detection: collision below + no destructible at that tile
           if player.level_map.check_collision(player.x, player.y + 1, player.w, player.h):
               if player.level_map.get_destructible_at(player.x, player.y + 1, player.w, player.h) is None:
                   return TickResult(dx=dx, dy=dy, request_exit=True, exit_reason="solid_landing")

           # 5. Drill continues
           return TickResult(dx=dx, dy=dy)
       ```

    ```python
        # Continue plunging at POGO_INITIAL_DY (no re-clamp like drill — gravity may apply via player.apply_physics).
        # Pogo is short-lived; the bounce/land determines exit.

        # Check destructible directly below (skip CRACKED_V — drill territory)
        tile_coord = player.level_map.get_destructible_at(player.x, player.y, player.w, player.h)
        if tile_coord:
            tx, ty = tile_coord
            tile_type = player.level_map.get_tile(tx, ty)
            if tile_type != INTGRID_CRACKED_V:
                # Soft destructible — break it, bounce, no juice refund (D-20: free).
                if player.game:
                    player.game.on_block_destroyed(tx, ty, tile_type)
                player.level_map.remove_tile(tx, ty)
                if player.game:
                    player.game.spawn_explosion(tx * tuning.TILE_SIZE, ty * tuning.TILE_SIZE, EXPLOSION_SIZE_PX)
                # Bounce off

            # CRACKED_V tile — NOT pogo-eligible. Treat as solid: land, no bounce.

            return TickResult(dx=0.0, dy=POGO_BOUNCE_VELOCITY, request_exit=True, exit_reason="bounced")

        # Check solid (non-destructible) below — land
        if player.level_map.check_collision(player.x, player.y + 1, player.w, player.h):
            return TickResult(dx=0.0, dy=POGO_INITIAL_DY, request_exit=True, exit_reason="landed")

        # No contact yet — keep plunging
        return TickResult(dx=0.0, dy=POGO_INITIAL_DY)

    def _touching_enemy(self, player) -> bool:
        """Returns True if player AABB overlaps any alive enemy."""
        if not player.game:
            return False
        for enemy in player.game.enemies:
            if not enemy.is_alive:
                continue
            if (player.x < enemy.x + enemy.w and player.x + player.w > enemy.x
                    and player.y < enemy.y + enemy.h and player.y + player.h > enemy.y):
                return True
        return False

    def _damage_touched_enemy(self, player) -> None:
        """Apply POGO_DAMAGE to the first enemy intersecting player AABB."""
        if not player.game:
            return
        for enemy in player.game.enemies:
            if not enemy.is_alive:
                continue
            if (player.x < enemy.x + enemy.w and player.x + player.w > enemy.x
                    and player.y < enemy.y + enemy.h and player.y + player.h > enemy.y):
                # Use existing enemy.take_damage if present, otherwise direct decrement.

                    enemy.take_damage(POGO_DAMAGE)

                return
    ```
